 44class DoorDetectionNode:
 45    """
 46    Own ROS interfaces and drive detection for synchronized camera frames.
 47
 48    The node caches camera intrinsics, converts ROS image encodings, refreshes a
 49    shared :class:`FrameContext`, advances the detector state machine, and
 50    publishes navigation results and optional debug images.
 51
 52    Attributes:
 53        latest_info:
 54            First received color-camera ``CameraInfo`` message.
 55
 56        fx:
 57            Cached horizontal focal length in pixels.
 58
 59        fy:
 60            Cached vertical focal length in pixels.
 61
 62        cx:
 63            Cached horizontal principal point.
 64
 65        cy:
 66            Cached vertical principal point.
 67
 68        go_to_idle_from_finish_state:
 69            Reset request forwarded into the frame context. Cleared once the
 70            machine has moved from the final state to idle.
 71
 72        start_door_frame_detection:
 73            Start request forwarded into the frame context. Cleared once the
 74            machine has left idle, so each request starts one detection cycle.
 75
 76        bridge:
 77            ``CvBridge`` used for ROS/OpenCV image conversion.
 78
 79        enable_visualization:
 80            Whether debug images and OpenCV drawing remain enabled.
 81
 82        sm:
 83            Registered frame-driven detection state machine.
 84
 85        door_detection_result_pub:
 86            Latched publisher for :class:`DoorDetectionResult`. It carries a
 87            finished cycle's result once, and a reset once the machine
 88            returns to idle.
 89
 90        color_branch_color_image_pub:
 91            Publisher for the color-branch debug image.
 92
 93        depth_branch_color_image_pub:
 94            Publisher for the depth-branch debug image.
 95
 96        viz_color_pub:
 97            Publisher for the stacked color-processing visualization.
 98
 99        viz_depth_pub:
100            Publisher for the stacked depth-processing visualization.
101
102        viz_plane_pub:
103            Publisher for the detected-plane overlay.
104
105        color_branch_color_image_annotated_pub:
106            ``/annotated`` twin of ``color_branch_color_image_pub``. Each debug
107            publisher has such a twin that receives only frames on which
108            something was drawn; the image recorder subscribes to these.
109
110        _frame_lock:
111            Lock that makes a frame arriving during processing be dropped
112            rather than queued.
113
114        dropped_frame_count:
115            Number of frames dropped because processing was still busy.

118    def __init__(self):
119        """
120        Configure ROS interfaces and construct the detection state machine.
121
122        Topic names, synchronization queue size, synchronization tolerance,
123        visualization, and profiling behavior are read from private ROS
124        parameters.
125
126        Raises:
127            KeyError:
128                If a required parameter without a default, such as
129                ``~profiler_output_path`` or ``~result_log_path``, is missing.
130
131        Notes:
132            Camera intrinsics are received separately from the synchronized
133            color/depth pair and must arrive before frame processing can begin.
134            With ``~profiling_enabled`` a pyinstrument session is started on
135            the first frame callback and saved after fifty frames, and stage
136            timings are flushed every five seconds. The image subscribers use a
137            queue of one so a slow node never processes a backlog of stale
138            frames.
139        """
140        # --- profiling setup ---
141        self.profile_enabled = rospy.get_param("~profiling_enabled", False)
142        self.profiler_output_path = rospy.get_param("~profiler_output_path")
143        self.profile_frame_count = 0
144        self.profile_frames = 50          # captures 50 synchronized frames then saves
145        self.profile_done = False
146        self.profiler = None
147        # pyinstrument requires that a session is started and stopped on the same
148        # thread, and all real work happens on the subscriber callback thread. The
149        # profiler is therefore created here but started lazily on the first
150        # callback, so both calls land on the thread being measured.
151        self.profile_started = False
152        if self.profile_enabled:
153            self.profiler = Profiler()
154            rospy.on_shutdown(self._save_profile)
155        # Timing averages are written on a timer instead of once per frame: the
156        # writer does a full read-modify-write of the output file, which is far too
157        # expensive to run at frame rate on flash storage.
158        if self.profile_enabled:
159            rospy.Timer(rospy.Duration(5.0), self._write_durations)
160                 
161
162        color_topic = rospy.get_param("~color_topic", "/camera/color/image_raw")
163        depth_topic = rospy.get_param("~depth_topic", "/camera/aligned_depth_to_color/image_raw")
164        info_topic = rospy.get_param("~camera_info_topic", "/camera/color/camera_info")
165        queue_size = rospy.get_param("~queue_size", 30)
166        slop = rospy.get_param("~sync_slop", 0.2)
167        self.sync_count = 0
168        # Serializes frame processing so a frame arriving mid-computation is dropped
169        # rather than queued behind the one in flight.
170        self._frame_lock = threading.Lock()
171        self.dropped_frame_count = 0
172        # Per-frame pair logging is a diagnostic; it publishes to /rosout every frame.
173        self.verbose_frame_logging = rospy.get_param("~verbose_frame_logging", False)
174        self.latest_info = None
175        self.fx = self.fy = self.cx = self.cy = None
176        self.go_to_idle_from_finish_state = False
177        self.start_door_frame_detection = False
178
179        self.bridge = CvBridge()
180        # Global visualization toggle (default true)
181        self.enable_visualization = rospy.get_param("~enable_visualization", True)
182        # Setup cv2 drawing to no-op when disabled
183        setup_visualization_mode(self.enable_visualization)
184        # The states form a gate: plane confirmation comes before frame lines,
185        # then fusion holds the result until the traversal side acknowledges it.
186        self.sm = StateMachine(ctx=None)
187        self.sm.add_state(idle_state())
188        self.sm.add_state(searching_door_plane_state())
189        self.sm.add_state(full_image_passability_check_state())
190        self.sm.add_state(dual_branch_frame_detection_state())
191        self.sm.add_state(combine_door_state())
192        self.sm.add_state(final_state())  # terminal state
193        self.sm.set_state("idle_state")
194
195        # state Publishers
196        # One message per cycle carries every value the passability node needs.
197        # Separate topics can arrive in any order, and latching lets a node that
198        # (re)connects late still receive the current result.
199        self.door_detection_result_pub = rospy.Publisher(
200            "~door_detection_result", DoorDetectionResult, queue_size=1, latch=True
201        )
202        # visualization publishers
203        self.color_branch_color_image_pub = rospy.Publisher("~viz/color_branch_color_image", Image, queue_size=1)
204        self.depth_branch_color_image_pub = rospy.Publisher("~viz/depth_branch_color_image", Image, queue_size=1)
205        self.viz_color_pub = rospy.Publisher("~viz/color_branch", Image, queue_size=1)
206        self.viz_depth_pub = rospy.Publisher("~viz/depth_branch", Image, queue_size=1)
207        self.viz_plane_pub = rospy.Publisher("~viz/plane_overlay", Image, queue_size=1)
208        # The same images, but only on frames where something was drawn on them.
209        # Image recording subscribes to these so plain camera frames are not saved.
210        self.color_branch_color_image_annotated_pub = rospy.Publisher("~viz/color_branch_color_image/annotated", Image, queue_size=1)
211        self.depth_branch_color_image_annotated_pub = rospy.Publisher("~viz/depth_branch_color_image/annotated", Image, queue_size=1)
212        self.viz_color_annotated_pub = rospy.Publisher("~viz/color_branch/annotated", Image, queue_size=1)
213        self.viz_depth_annotated_pub = rospy.Publisher("~viz/depth_branch/annotated", Image, queue_size=1)
214        self.viz_plane_annotated_pub = rospy.Publisher("~viz/plane_overlay/annotated", Image, queue_size=1)
215        
216
217        # Subscribers: sync color + depth; cache camera info separately for robustness
218        # queue_size=1 is essential: rospy defaults to an unbounded receive queue, so a
219        # consumer slower than the camera would buffer full-resolution images without
220        # limit and act on ever more stale frames. buff_size must exceed one image or
221        # each frame costs many socket reads.
222        color_sub = message_filters.Subscriber(color_topic, Image, queue_size=1, buff_size=2**22)
223        depth_sub = message_filters.Subscriber(depth_topic, Image, queue_size=1, buff_size=2**22)
224        
225        # camera intrinsics will be cached on first receipt
226        rospy.Subscriber(info_topic, CameraInfo, self._info_cb, queue_size=10)
227        rospy.Subscriber("/check_if_corridor_is_passable/retrigger_door_detection_node", Bool, self._retrigger_door_detection_node_cb, queue_size=1)
228        
229        # published by top level path planning and navigation algorithm to trigger start of door frame detection and subsequent steps. Published once per door traversal attempt.
230        rospy.Subscriber("/trigger_start_door_frame_detection", Bool, self._trigger_start_door_frame_detection, queue_size=1)
231
232        # Exact timestamps differ between several camera drivers, so tolerate a
233        # small skew while still processing color and depth as one observation.
234        ats = message_filters.ApproximateTimeSynchronizer(
235            [color_sub, depth_sub], queue_size=queue_size, slop=slop
236        )
237        ats.registerCallback(self.callback)

253    def _stop_profile(self):
254        """
255        End the sampling session and write the HTML report.
256
257        Notes:
258            Must run on the thread that started the session. Called from the
259            frame callback once enough frames are captured. Any failure is logged
260            rather than raised so profiling cannot break detection.
261        """
262        if self.profiler is None or self.profile_done:
263            return
264        self.profile_done = True
265        try:
266            if self.profile_started:
267                self.profiler.stop()
268                with open(self.profiler_output_path, "w") as f:
269                    f.write(self.profiler.output_html())
270                # A plain-text copy is far easier to read over a terminal session
271                # on the robot than the HTML report.
272                text_path = os.path.splitext(self.profiler_output_path)[0] + ".txt"
273                with open(text_path, "w") as f:
274                    f.write(self.profiler.output_text(unicode=False, color=False))
275                rospy.loginfo("[Profiler] Saved to %s and %s",
276                              self.profiler_output_path, text_path)
277        except Exception as exc:
278            rospy.logwarn("[Profiler] Could not save profile: %s", exc)

377    def _publish_detection_result(self, ctx, stamp):
378        """
379        Publish the frame context's door result as one latched message.
380
381        Args:
382            ctx:
383                Shared frame context holding the cycle's outputs.
384
385            stamp:
386                Timestamp of the camera frame that produced them.
387
388        Notes:
389            Called on the transition into the final state, and on the return
390            to idle, where the cleared context yields the reset message.
391            Missing values use the message's sentinels: ``"unknown"`` for the
392            label, NaN for the depth and plane distance, ``-1`` for the pixel,
393            and a zero plane normal. Only the reset lacks a plane: a finished
394            cycle always has one, either confirmed or, for
395            ``No_door_plane_detected``, the reference distance with normal
396            ``(0, 0, 1)``.
397        """
398        msg = DoorDetectionResult()
399        msg.header.stamp = stamp
400        msg.door_state = ctx.door_state_label if ctx.door_state_label is not None else "unknown"
401        msg.door_depth = float(ctx.door_depth) if ctx.door_depth is not None else float("nan")
402        mfx = ctx.mid_frame_x_px_for_passability_check
403        msg.mid_frame_x_px = int(mfx) if mfx is not None else -1
404        if ctx.plane_result is not None:
405            msg.plane_distance = float(ctx.plane_result["distance_m"])
406            normal = ctx.plane_result["plane_norm_vector"]
407            msg.plane_normal.x = float(normal[0])
408            msg.plane_normal.y = float(normal[1])
409            msg.plane_normal.z = float(normal[2])
410        else:
411            msg.plane_distance = float("nan")
412        self.door_detection_result_pub.publish(msg)
413        rospy.loginfo("Published door detection result: %s", msg.door_state)

463    def _to_depth_mm_and_m(self, depth_msg: Image):
464        """
465        Convert one ROS depth message into millimeter and meter arrays.
466
467        Args:
468            depth_msg:
469                Depth image encoded as uint16 millimeters or floating-point
470                meters.
471
472        Returns:
473            Tuple ``(depth_mm, depth_m)`` containing uint16 millimeters and
474            float32 meters.
475
476        Notes:
477            Floating-point NaN and infinity values are replaced with zero only
478            for the uint16 adapter image. The metric float image retains the
479            original nonfinite values for later validity masking.
480
481            The millimetre image feeds only the RealSense-compatible adapter used
482            by the diagnostic views, so it is produced only when visualization is
483            enabled; otherwise ``None`` is returned in its place.
484        """
485        img = self.bridge.imgmsg_to_cv2(depth_msg, desired_encoding="passthrough")
486        # Keep both units: geometry is easier to reason about in metres, while
487        # the RealSense-compatible adapter expects the original millimetres.
488        if depth_msg.encoding in ("16UC1", "mono16"):
489            # astype() always copies, and the array is already uint16 here, so
490            # convert straight to metres and scale by a reciprocal.
491            depth_m = img.astype(np.float32) * np.float32(0.001)
492            depth_mm = img.astype(np.uint16, copy=False) if self.enable_visualization else None
493        else:
494            depth_m = img.astype(np.float32)
495            if self.enable_visualization:
496                # Sanitize invalid values before casting (NaN/Inf can trigger warnings)
497                scaled_mm = depth_m * 1000.0
498                scaled_mm = np.nan_to_num(scaled_mm, nan=0.0, posinf=0.0, neginf=0.0)
499                depth_mm = np.clip(np.rint(scaled_mm), 0, 65535).astype(np.uint16)
500            else:
501                depth_mm = None
502        return depth_mm, depth_m

568    def callback(self, color_msg: Image, depth_msg: Image):
569        """
570        Process one approximately synchronized color/depth message pair.
571
572        If a previous pair is still being processed, this pair is dropped and
573        counted; otherwise it is handed to :meth:`_process_frame`.
574
575        Args:
576            color_msg:
577                Color image selected by ``ApproximateTimeSynchronizer``.
578
579            depth_msg:
580                Aligned depth image paired with ``color_msg``.
581
582        Notes:
583            Dropping instead of queueing keeps the published door state tied
584            to a recent observation when detection is slower than the camera.
585        """
586        # Drop frames that arrive while a previous one is still being processed.
587        # Without this the node would fall progressively further behind the camera
588        # and publish a door state describing an increasingly old observation.
589        if not self._frame_lock.acquire(blocking=False):
590            self.dropped_frame_count += 1
591            rospy.logwarn_throttle(
592                5.0,
593                "Detection is slower than the camera; dropped %d frame(s) so far.",
594                self.dropped_frame_count,
595            )
596            return
597        try:
598            self._process_frame(color_msg, depth_msg)
599        finally:
600            self._frame_lock.release()
601
602    def _process_frame(self, color_msg: Image, depth_msg: Image):
603        """
604        Run one detection step for an already-admitted frame pair.
605
606        The frame is converted, the shared context is created or refreshed, the
607        state machine advances one step, and the navigation results and
608        optional debug images are published.
609
610        Args:
611            color_msg:
612                Color image selected by ``ApproximateTimeSynchronizer``.
613
614            depth_msg:
615                Aligned depth image paired with ``color_msg``.
616
617        Notes:
618            Split out of :meth:`callback` so the drop-if-busy guard wraps the
619            whole body. Processing is deferred until camera intrinsics are
620            available. Per-branch color copies and the depth-frame adapter are
621            created only when visualization is enabled; otherwise the branches
622            share the incoming frame and ``ctx.depth_frame`` is ``None``.
623
624            The start request is cleared when the step leaves idle and the
625            idle request when the step moves from the final state to idle.
626
627            The door result is published once, on ``~door_detection_result``,
628            when the step enters the final state, and a reset once when it
629            returns to idle. Visualization errors are logged at debug level
630            and never stop the navigation outputs.
631        """
632        # Start sampling on this thread, which is where the work happens and the
633        # only thread the session may later be stopped from.
634        if self.profile_enabled and not self.profile_started and not self.profile_done:
635            self.profiler.start()
636            self.profile_started = True
637
638        self.sync_count += 1
639        if self.verbose_frame_logging:
640            rospy.loginfo(
641                "pair=%d color=%.6f depth=%.6f delta=%.6f thread=%s",
642                self.sync_count,
643                color_msg.header.stamp.to_sec(),
644                depth_msg.header.stamp.to_sec(),
645                abs((color_msg.header.stamp - depth_msg.header.stamp).to_sec()),
646                threading.current_thread().name,
647            )
648        color_image = self._to_cv_color(color_msg)
649        depth_mm, depth_m = self._to_depth_mm_and_m(depth_msg)
650        # Use latest camera info; require not None
651        # Running geometry with guessed intrinsics would yield plausible-looking
652        # but physically wrong plane sizes, so wait instead.
653        if self.fx is None:
654            rospy.logwarn_throttle(5.0, "Waiting for CameraInfo (only needed once)...")
655            return
656
657        # The adapter exists purely to satisfy the RealSense-style access pattern
658        # used by the diagnostic views, so build it only when those run.
659        depth_frame_adapter = (
660            DepthFrameAdapter(depth_mm, self.fx, self.fy, self.cx, self.cy)
661            if self.enable_visualization
662            else None
663        )
664
665        # Reuse the context to preserve state outputs, but give each detector its
666        # own image copy because overlays are drawn in place.
667        if not hasattr(self.sm, "ctx") or self.sm.ctx is None: # create object of context class once.
668            self.sm.ctx = FrameContext(
669                depth_image_in_meters=depth_m,
670                color_image=color_image,
671                fx=self.fx, fy=self.fy, cx=self.cx, cy=self.cy,
672                depth_frame=depth_frame_adapter,
673            )
674            self._assign_branch_wise_debug_images(self.sm.ctx, color_image)
675            self._seed_runtime_door_geometry(self.sm.ctx)
676            self.sm.ctx.go_to_idle_from_finish_state = False
677            self.sm.ctx.start_door_frame_detection = False
678        else: # update existing context object with new subscriped data.
679            ctx = self.sm.ctx
680            ctx.depth_image_in_meters = depth_m
681            ctx.color_image = color_image
682            self._assign_branch_wise_debug_images(ctx, color_image)
683            ctx.fx, ctx.fy, ctx.cx, ctx.cy = self.fx, self.fy, self.cx, self.cy
684            ctx.depth_frame = depth_frame_adapter
685            ctx.go_to_idle_from_finish_state = self.go_to_idle_from_finish_state
686            ctx.start_door_frame_detection = self.start_door_frame_detection
687
688        
689        # One synchronized frame drives exactly one state-machine step.
690        state_before = self.sm.current_state.name
691        self.sm.update(self.sm.ctx)
692        state_after = self.sm.current_state.name
693        # Each request arrives as a single True message, so it is consumed on the
694        # transition it caused. Otherwise it would stay set and act again in the
695        # next detection cycle.
696        if state_before == "idle_state" and state_after != "idle_state":
697            self.start_door_frame_detection = False
698        if state_before == "final_state" and state_after == "idle_state":
699            self.go_to_idle_from_finish_state = False
700        # The result is sent once when a cycle finishes, and the reset once when
701        # the machine goes back to idle; the idle entry hook has already cleared
702        # the context, so the same builder produces the reset.
703        if state_after == "final_state" and state_before != "final_state":
704            self._publish_detection_result(self.sm.ctx, color_msg.header.stamp)
705        if state_before == "final_state" and state_after == "idle_state":
706            self._publish_detection_result(self.sm.ctx, color_msg.header.stamp)
707        # Timing averages are flushed by a timer, not here: the writer rewrites the
708        # whole output file and must not run at frame rate.
709        # Stop sampling once enough frames are captured so the report stays bounded.
710        if self.profile_enabled and not self.profile_done:
711            self.profile_frame_count += 1
712            if self.profile_frame_count >= self.profile_frames:
713                self._stop_profile()
714                rospy.loginfo("[Profiler] Captured %d frames.", self.profile_frames)
715
716       
717
718        
719
720        # Visualization failures must never stop the navigation-facing outputs above.
721        # Publish visualizations
722        try:
723            if self.enable_visualization:
724                ctx = self.sm.ctx
725                # The stacks are built from the branch images, so they follow the
726                # same test. That also keeps a stack left over from an earlier
727                # frame, which is republished while other states run, off the
728                # annotated topics.
729                color_has_info = self._has_diagnostics(ctx.color_image_color_based, ctx.banner_color_based)
730                depth_has_info = self._has_diagnostics(ctx.color_image_depth_based, ctx.banner_depth_based)
731                if ctx.color_image_color_based is not None:
732                    color_branch_vis = add_banner(ctx.color_image_color_based, ctx.banner_color_based)
733                    self._publish_viz(self.color_branch_color_image_pub, self.color_branch_color_image_annotated_pub, color_branch_vis, color_has_info)
734                if ctx.color_image_depth_based is not None:
735                    depth_branch_vis = add_banner(ctx.color_image_depth_based, ctx.banner_depth_based)
736                    self._publish_viz(self.depth_branch_color_image_pub, self.depth_branch_color_image_annotated_pub, depth_branch_vis, depth_has_info)
737                if ctx.viz_color_stack is not None:
738                    self._publish_viz(self.viz_color_pub, self.viz_color_annotated_pub, ctx.viz_color_stack, color_has_info)
739                if ctx.viz_depth_stack is not None:
740                    self._publish_viz(self.viz_depth_pub, self.viz_depth_annotated_pub, ctx.viz_depth_stack, depth_has_info)
741                # plane overlay: use base color image (with plane outlines drawn)
742                if ctx.color_image_for_plane_detection is not None:
743                    plane_vis = add_banner(ctx.color_image_for_plane_detection, ctx.banner_for_plane_detection)
744                    plane_has_info = self._has_diagnostics(ctx.color_image_for_plane_detection, ctx.banner_for_plane_detection)
745                    self._publish_viz(self.viz_plane_pub, self.viz_plane_annotated_pub, plane_vis, plane_has_info)
746
747        except Exception as e:
748            rospy.logdebug(f"Viz publish exception: {e}")
749
750        #cv2.waitKey(1)

123        Topic names, synchronization queue size, synchronization tolerance,

125        parameters.

127        Raises:
128            KeyError:

130                ``~profiler_output_path`` or ``~result_log_path``, is missing.

138            queue of one so a slow node never processes a backlog of stale

142        self.profile_enabled = rospy.get_param("~profiling_enabled", False)

148        # pyinstrument requires that a session is started and stopped on the same
149        # thread, and all real work happens on the subscriber callback thread. The
150        # profiler is therefore created here but started lazily on the first

378    def _publish_detection_result(self, ctx, stamp):
379        """
380        Publish the frame context's door result as one latched message.
381
382        Args:
383            ctx:
384                Shared frame context holding the cycle's outputs.
385
386            stamp:
387                Timestamp of the camera frame that produced them.
388
389        Notes:
390            Called on the transition into the final state, and on the return
391            to idle, where the cleared context yields the reset message.
392            Missing values use the message's sentinels: ``"unknown"`` for the
393            label, NaN for the depth and plane distance, ``-1`` for the pixel,
394            and a zero plane normal. Only the reset lacks a plane: a finished
395            cycle always has one, either confirmed or, for
396            ``No_door_plane_detected``, the reference distance with normal
397            ``(0, 0, 1)``.
398        """
399        msg = DoorDetectionResult()
400        msg.header.stamp = stamp
401        msg.door_state = ctx.door_state_label if ctx.door_state_label is not None else "unknown"
402        msg.door_depth = float(ctx.door_depth) if ctx.door_depth is not None else float("nan")
403        mfx = ctx.mid_frame_x_px_for_passability_check
404        msg.mid_frame_x_px = int(mfx) if mfx is not None else -1
405        if ctx.plane_result is not None:
406            msg.plane_distance = float(ctx.plane_result["distance_m"])
407            normal = ctx.plane_result["plane_norm_vector"]
408            msg.plane_normal.x = float(normal[0])
409            msg.plane_normal.y = float(normal[1])
410            msg.plane_normal.z = float(normal[2])
411        else:
412            msg.plane_distance = float("nan")
413        self.door_detection_result_pub.publish(msg)
414        rospy.loginfo("Published door detection result: %s", msg.door_state)

465        """
466        Convert one ROS depth message into millimeter and meter arrays.
467
468        Args:
469            depth_msg:
470                Depth image encoded as uint16 millimeters or floating-point
471                meters.
472
473        Returns:
474            Tuple ``(depth_mm, depth_m)`` containing uint16 millimeters and
475            float32 meters.

588        # Without this the node would fall progressively further behind the camera
589        # and publish a door state describing an increasingly old observation.
590        if not self._frame_lock.acquire(blocking=False):
591            self.dropped_frame_count += 1

594                "Detection is slower than the camera; dropped %d frame(s) so far.",
595                self.dropped_frame_count,

597            return
598        try:
599            self._process_frame(color_msg, depth_msg)
600        finally:
601            self._frame_lock.release()

603    def _process_frame(self, color_msg: Image, depth_msg: Image):
604        """
605        Run one detection step for an already-admitted frame pair.
606
607        The frame is converted, the shared context is created or refreshed, the
608        state machine advances one step, and the navigation results and
609        optional debug images are published.
610
611        Args:
612            color_msg:
613                Color image selected by ``ApproximateTimeSynchronizer``.
614
615            depth_msg:
616                Aligned depth image paired with ``color_msg``.
617
618        Notes:
619            Split out of :meth:`callback` so the drop-if-busy guard wraps the
620            whole body. Processing is deferred until camera intrinsics are
621            available. Per-branch color copies and the depth-frame adapter are
622            created only when visualization is enabled; otherwise the branches
623            share the incoming frame and ``ctx.depth_frame`` is ``None``.
624
625            The start request is cleared when the step leaves idle and the
626            idle request when the step moves from the final state to idle.
627
628            The door result is published once, on ``~door_detection_result``,
629            when the step enters the final state, and a reset once when it
630            returns to idle. Visualization errors are logged at debug level
631            and never stop the navigation outputs.
632        """
633        # Start sampling on this thread, which is where the work happens and the
634        # only thread the session may later be stopped from.
635        if self.profile_enabled and not self.profile_started and not self.profile_done:

653        # but physically wrong plane sizes, so wait instead.
654        if self.fx is None:

702        # the machine goes back to idle; the idle entry hook has already cleared

706        if state_before == "final_state" and state_after == "idle_state":
707            self._publish_detection_result(self.sm.ctx, color_msg.header.stamp)
708        # Timing averages are flushed by a timer, not here: the writer rewrites the
709        # whole output file and must not run at frame rate.
710        # Stop sampling once enough frames are captured so the report stays bounded.
711        if self.profile_enabled and not self.profile_done:
712            self.profile_frame_count += 1
713            if self.profile_frame_count >= self.profile_frames:
714                self._stop_profile()
715                rospy.loginfo("[Profiler] Captured %d frames.", self.profile_frames)
716

718
719

721        # Visualization failures must never stop the navigation-facing outputs above.
722        # Publish visualizations
723        try:

741                    self._publish_viz(self.viz_depth_pub, self.viz_depth_annotated_pub, ctx.viz_depth_stack, depth_has_info)
