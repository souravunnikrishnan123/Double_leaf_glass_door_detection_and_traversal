 14class TemporalSmoother:
 15    """
 16    Smooth categorical door states with plurality voting and hysteresis.
 17
 18    A bounded window supplies the most frequent label as the candidate. No
 19    output is produced until one label appears ``min_consistent`` times, and
 20    an established label is kept until a different label both leads the window
 21    and reaches that count.
 22
 23    Attributes:
 24        window_size:
 25            Maximum number of recent labels retained.
 26
 27        min_consistent:
 28            Minimum count the leading label needs, within the window, to
 29            become the first stable label or to replace the current one.
 30
 31        hysteresis:
 32            Whether switching away from the stable label is additionally
 33            guarded by ``stable_hold``.
 34
 35        stable_hold:
 36            Minimum count the leading label needs before the guard lets it
 37            replace the stable label. It only has an effect when it exceeds
 38            ``min_consistent``.
 39
 40        buffer:
 41            Bounded deque of recent labels.
 42
 43        current_stable:
 44            Label currently exposed to downstream states, or ``None`` during
 45            warm-up.
 46
 47        stable_count:
 48            Confidence-like count maintained for the current stable label. It
 49            is informational only and does not influence the returned label.
 50    """
 51
 52    def __init__(self, window_size=10, min_consistent=3, hysteresis=True, stable_hold=2):
 53        """
 54        Initialize an empty categorical smoothing window.
 55
 56        Args:
 57            window_size:
 58                Maximum retained labels.
 59
 60            min_consistent:
 61                Minimum repeated-label count required during warm-up and
 62                candidate selection.
 63
 64            hysteresis:
 65                Enable guarded switching between stable labels.
 66
 67            stable_hold:
 68                Minimum candidate count required before switching. Values not
 69                greater than ``min_consistent`` add no extra guard, because a
 70                candidate already needs ``min_consistent`` occurrences.
 71        """
 72        self.window_size = window_size
 73        self.min_consistent = min_consistent
 74        self.hysteresis = hysteresis
 75        self.stable_hold = stable_hold
 76        self.buffer = deque(maxlen=window_size)
 77        self.current_stable = None
 78        self.stable_count = 0
 79
 80    def reset(self) -> None:
 81        """Discard all history so the next observation starts a fresh warm-up."""
 82        self.buffer.clear()
 83        self.current_stable = None
 84        self.stable_count = 0
 85
 86    def update(self, label: str) -> Optional[str]:
 87        """
 88        Add one observation and return the current stable label.
 89
 90        Args:
 91            label:
 92                Current fused door-state observation.
 93
 94        Returns:
 95            Stable label, or ``None`` while the initial window lacks
 96            ``min_consistent`` evidence.
 97
 98        Notes:
 99            The candidate is the most frequent label in the window, not
100            necessarily an absolute majority. Ties go to the label that first
101            appears in the window, i.e. the oldest one. With hysteresis
102            disabled, every sufficiently supported candidate replaces the
103            current stable label immediately. With the settings used by
104            :class:`combine_door_state` (``min_consistent=3``,
105            ``stable_hold=2``) both modes behave the same.
106        """
107        # Keep raw categorical labels; averaging numeric encodings would invent
108        # states that have no physical meaning.
109        # Push new value
110        self.buffer.append(label)
111
112        # Count frequencies over window
113        freq = defaultdict(int)
114        for v in self.buffer:
115            freq[v] += 1
116
117        # Majority vote
118        majority_label = max(freq.items(), key=lambda x: x[1])[0]
119        majority_count = freq[majority_label]
120
121        # Initial warm-up: until enough evidence, return None
122        if self.current_stable is None:
123            if majority_count >= self.min_consistent:
124                self.current_stable = majority_label
125                self.stable_count = majority_count
126                return self.current_stable
127            else:
128                return None
129
130        # Require at least min_consistent within window
131        candidate = majority_label if majority_count >= self.min_consistent else self.current_stable
132
133        if not self.hysteresis:
134            self.current_stable = candidate
135            self.stable_count = majority_count
136            return self.current_stable
137
138        # Hysteresis keeps a single bad depth frame from flipping an already
139        # published navigation decision.
140        # Hysteresis: only switch after "stable_hold" confirmations for new candidate
141        if candidate == self.current_stable:
142            # reinforce stability
143            self.stable_count = min(self.window_size, self.stable_count + 1)
144            return self.current_stable
145        else:
146            # moving toward a new state: require repeated confirmations
147            if majority_count >= self.stable_hold:
148                self.current_stable = candidate
149                self.stable_count = majority_count
150            else:
151                # keep previous until enough evidence
152                self.stable_count = max(0, self.stable_count - 1)
153            return self.current_stable

155class combine_door_state(BaseState):
156    """
157    Resolve detector-branch disagreements and route the state machine.
158
159    A fixed decision table chooses a label, source pipeline, and door depth.
160    Matching open-side polygons are compared with intersection-over-union.
161    The selected label is then temporally smoothed before navigation outputs
162    are written to the frame context. An open result is published only
163    together with geometry from the current frame, and a ``no_frame_detected``
164    result only after it has persisted for a configurable number of frames.
165
166    Attributes:
167        height:
168            Current branch image height used to rasterize ROI masks.
169
170        width:
171            Current branch image width used to rasterize ROI masks.
172
173        smoother:
174            :class:`TemporalSmoother` applied to fused labels.
175
176        smoothed_door_state_log_path:
177            File receiving one smoothed label per frame when result logging is
178            enabled.
179
180        enable_result_log:
181            Value of ``~enable_result_log``.
182
183        _result_log_file:
184            Append-mode handle kept open for the node lifetime, or ``None``
185            when logging is disabled or the file could not be opened.
186
187        number_of_times_no_frame_detected_as_smoothed_door_state:
188            Consecutive-frame counter for the smoothed
189            ``"no_frame_detected"`` label; reset whenever the smoothed label
190            changes to something else.
191
192        maximum_allowed_number_of_times_no_frame_detected_as_smoothed_door_state:
193            Consecutive-frame threshold the counter above must reach before
194            ``"no_frame_detected"`` is treated as final and escalated to
195            ``"full_image_passability_check_state"``.

243
244        Args:
245            line:
246                Text to write, including its trailing newline.

274    def rois_match(self, roi1, roi2, iou_threshold):
275        """
276        Check polygon agreement using image-space intersection over union.
277
278        Args:
279            roi1:
280                First ``N x 2`` polygon, or ``None``.
281
282            roi2:
283                Second ``N x 2`` polygon, or ``None``.
284
285            iou_threshold:
286                Minimum intersection-over-union required for a match.
287
288        Returns:
289            ``True`` when both polygons have nonzero union and meet the
290            threshold; otherwise ``False``.
291
292        Notes:
293            :attr:`height` and :attr:`width` must be set for the current image
294            before this method is called.
295        """
296        mask1 = np.zeros((self.height, self.width), dtype=np.uint8)
297        mask2 = np.zeros((self.height, self.width), dtype=np.uint8)
298
299        if roi1 is None or roi2 is None:
300            return False
301
302        # Raster masks make IoU work for arbitrary quadrilaterals, including
303        # the slanted ROIs produced by perspective.
304        cv2.fillPoly(mask1, [roi1.astype(np.int32)], 255)
305        cv2.fillPoly(mask2, [roi2.astype(np.int32)], 255)
306
307        intersection = np.logical_and(mask1, mask2).sum()
308        union = np.logical_or(mask1, mask2).sum()
309
310        if union == 0:
311            return False
312
313        iou= intersection / union
314        return iou >= iou_threshold

319        """
320        Apply the color/depth branch-fusion decision table.
321
322        Depth detections are trusted when color is weak, while valid color
323        detections fill gaps when depth finds no frame or is unknown. Direct
324        conflicts fall back to an unknown/full-image result. Matching open
325        labels must also agree spatially; the color pipeline is then selected.
326        ``"no_frame_detected"`` is produced only when both branches report it.
327
328        Args:
329            color_pipline_result:
330                Color-branch label.
331
332            depth_pipline_result:
333                Depth-branch label.
334
335            roi_open_side_color_based:
336                Color-branch opening polygon, if any.
337
338            roi_open_side_depth_based:
339                Depth-branch opening polygon, if any.
340
341            door_depth_m_color_based:
342                Color-branch door-depth estimate.
343
344            door_depth_m_depth_based:
345                Depth-branch door-depth estimate.
346
347            iou_threshold:
348                ROI agreement threshold for matching open labels.
349
350        Returns:
351            A dictionary with the final label, trusted pipeline, selected door
352            depth, escalation flag, and a human-readable reason.
353
354        Notes:
355            The dictionary always contains ``final_door_status``, ``pipeline``,
356            ``ask_human``, ``door_depth``, and ``reason``. ``pipeline`` is
357            ``"color_based"``, ``"depth_based"``, ``"full_image_view"``, or
358            ``None`` for a label combination the table does not list; the last
359            case keeps the default ``"unknown"`` result.
360        """
361
362        # Start conservatively. Every trusted combination below must opt into a
363        # pipeline and a usable door depth.
364        # Default result container
365        result = dict(final_door_status="unknown", pipeline=None, ask_human=False, door_depth = None, reason="")

379                result["door_depth"] = door_depth_m_color_based # use color-based door depth
380                result["reason"] = "Color closed + depth no-frame → closed"
381            elif color_pipline_result == "no_frame_detected":
382                result["final_door_status"] = "no_frame_detected"
383                result["pipeline"] = "full_image_view"
384                result["ask_human"] = True
385                result["door_depth"] = None
386                result["reason"] = "No frame in both → no_frame_detected, ask human/full image"
387            elif color_pipline_result == "unknown":
388                result["final_door_status"] = "unknown"
389                result["pipeline"] = "full_image_view"
390                result["ask_human"] = True
391                result["door_depth"] = None
392                result["reason"] = "Depth no-frame + color unknown → unknown"
393
394        # -------------------------------------------------------------
395        # DEPTH = UNKNOWN

397        elif depth_pipline_result == "unknown":

403            elif color_pipline_result == "closed":

407                result["reason"] = "Color closed + depth unknown → closed"

509    def do_action(self, ctx: FrameContext) -> Optional[str]:
510        """
511        Fuse one frame, update navigation outputs, and select the next state.
512
513        Args:
514            ctx:
515                Shared frame context containing both branch results and ROIs.
516
517        Returns:
518            Next state name. Open labels with agreeing geometry go to
519            ``"final_state"``; a smoothed no-frame label goes to
520            ``"full_image_passability_check_state"`` only after
521            ``maximum_allowed_number_of_times_no_frame_detected_as_smoothed_door_state``
522            consecutive no-frame frames, staying in
523            ``"dual_branch_frame_detection_state"`` while still waiting;
524            other labels also return to ``"dual_branch_frame_detection_state"``.
525
526        Notes:
527            ``ctx.door_depth`` and ``ctx.mid_frame_x_px_for_passability_check``
528            are cleared on every call and set only on the transition to
529            ``"final_state"``. That transition requires the current raw label to
530            equal the smoothed open label and to supply both an open-side ROI
531            and a door depth; the geometry therefore always comes from the
532            current frame.
533
534            ``ctx.door_state_label`` is set on every call: the open label or
535            ``"no_frame_detected"`` only on the transitions above,
536            ``"closed"`` for a smoothed closed label, and ``"unknown"``
537            otherwise, including warm-up, an open label without usable
538            geometry, and the no-frame waiting period.
539
540            The smoother and the no-frame counter are reset when the detection
541            cycle ends, so the next cycle starts without earlier votes. The
542            counter is also reset whenever the smoothed label is not
543            ``"no_frame_detected"``.
544
545            For an open-left result the 95th ROI x-percentile marks the central
546            frame edge; open-right uses the 5th percentile. Diagnostic log
547            write failures are logged, not raised.
548        """
549        self.height, self.width = ctx.color_image_color_based.shape[:2]
550
551        # Label agreement is not enough for an open door: both branches must be
552        # talking about substantially the same image region.
553        result = self.resolve_door_status(ctx.door_state_color_based, ctx.door_state_depth_based, ctx.roi_open_side_color_based, ctx.roi_open_side_depth_based, ctx.door_depth_m_color_based, ctx.door_depth_m_depth_based, iou_threshold=0.5)
554
555        raw_door_state = result["final_door_status"]
556
557        # Apply temporal smoothing on final label
558        smoothed_door_state = self.smoother.update(raw_door_state)
559
560        if self.enable_result_log:
561            self._write_result_log(f"smoothed_door_state: {smoothed_door_state}\n")
562
563        ctx.door_depth = None
564        ctx.mid_frame_x_px_for_passability_check = None
565
566        if smoothed_door_state in ("open_left", "open_right"):
567            # Smoothing confirms the label only. ROI x and door depth are tied to the
568            # camera pose of the frame they were measured in, so they must come from
569            # the current frame, and only if it agrees with the smoothed label (a
570            # conflict frame has pipeline "full_image_view" and no ROI).
571            if raw_door_state == smoothed_door_state:
572                roi_open_side = getattr(ctx, f"roi_open_side_{result['pipeline']}") # get the open side roi polygon
573                if roi_open_side is not None and result["door_depth"] is not None:
574                    # Use an inner-edge percentile rather than a single polygon vertex; small
575                    # ROI rotations then have little effect on the corridor anchor.

278
279        Args:

282

285

291            threshold; otherwise ``False``.
292
293        Notes:
294            :attr:`height` and :attr:`width` must be set for the current image
295            before this method is called.
296        """
297        mask1 = np.zeros((self.height, self.width), dtype=np.uint8)
298        mask2 = np.zeros((self.height, self.width), dtype=np.uint8)
299
300        if roi1 is None or roi2 is None:
301            return False
302
303        # Raster masks make IoU work for arbitrary quadrilaterals, including
304        # the slanted ROIs produced by perspective.
305        cv2.fillPoly(mask1, [roi1.astype(np.int32)], 255)
306        cv2.fillPoly(mask2, [roi2.astype(np.int32)], 255)
307
308        intersection = np.logical_and(mask1, mask2).sum()
309        union = np.logical_or(mask1, mask2).sum()
310
311        if union == 0:
312            return False
313
314        iou= intersection / union
315        return iou >= iou_threshold
