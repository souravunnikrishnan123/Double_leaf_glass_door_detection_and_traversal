193    def remove_floor_ransac(self, points, uv):
194        """
195        Remove the first sufficiently large horizontal RANSAC plane.
196
197        Plane fitting stops when a camera-Y-aligned normal is found. Inlier
198        points belonging to that plane are removed while UV correspondence is
199        preserved.
200
201        Args:
202            points:
203                ``N x 3`` camera-frame point array.
204
205            uv:
206                ``N x 2`` image coordinates paired with ``points``.
207
208        Returns:
209            Tuple ``(non_floor_points, non_floor_uv, floor_height)``.
210            ``floor_height`` is the mean camera-frame Y of floor inliers, or
211            ``None`` when no horizontal plane is found.
212
213        Notes:
214            The method assumes the first accepted horizontal plane is the
215            floor. When fitting fails, all input points are retained.
216        """
217        #ransac
218        floor_inliers = None
219        for _ in range(self.max_planes):
220            if len(points) < self.ransac_n:
221                break
222
223            pc = o3d.geometry.PointCloud()
224            pc.points = o3d.utility.Vector3dVector(points)
225            plane_model, inliers = pc.segment_plane(
226                distance_threshold=self.distance_threshold,
227                ransac_n=self.ransac_n,
228                num_iterations=self.num_iterations,
229            )
230            if len(inliers) < self.min_inliers:
231                break
232
233            a, b, c, d = plane_model
234            # Check if the plane is horizontal
235            normal = np.array(plane_model[:3])
236            normal = normal / np.linalg.norm(normal)
237            if abs(abs(normal[1]) - 1.0) < self.horizontal_tol:
238                floor_inliers = np.array(inliers, dtype=int)
239                break
240
241        floor_height = None
242        if floor_inliers is not None and floor_inliers.size > 0:
243            #take the largest horizontal plane as floor. this is an assumption.

251            uv_3_nofloor = uv
252
253        if floor_height is None:
254            print("Detected floor height: None (RANSAC failed)")
255        else:
256            print(f"Detected floor height at y={floor_height:.3f} meters")
257            pass

274    def normal_filtering(self, points, uv):
275        """
276        Keep obstacle-like surfaces and preserve their pixel correspondence.
277
278        Args:
279            points:
280                ``N x 3`` floor-filtered point array.
281
282            uv:
283                ``N x 2`` pixel coordinates paired with ``points``.
284
285        Returns:
286            Tuple of points and UV coordinates whose estimated normals satisfy
287            ``abs(normal_y) < ny_thr``.
288
289        Notes:
290            If Open3D normal estimation fails, the original arrays are returned
291            unchanged.
292        """
293        # 4) Keep points whose normals are close to camera Z axis (door plane direction)
294        pc_nf = o3d.geometry.PointCloud()
295        pc_nf.points = o3d.utility.Vector3dVector(points)
296        try:
297            pc_nf.estimate_normals(search_param=o3d.geometry.KDTreeSearchParamHybrid(radius=self.radius, max_nn=self.max_nn))
298            pc_nf.orient_normals_towards_camera_location(np.array([0.0, 0.0, 0.0]))
299            normals = np.asarray(pc_nf.normals)
300        except Exception:
301            normals = None
302
303        if normals is None:
304            points_4_normal = points
305            uv_4_normal = uv
306        else:
307            vertical_mask = np.abs(normals[:, 1]) < self.ny_thr
308            points_4_normal = points[vertical_mask]
309            uv_4_normal = uv[vertical_mask]
310        return points_4_normal, uv_4_normal

312    def outlier_removal(self, points, uv):
313        """
314        Apply Open3D statistical outlier removal to paired points and pixels.
315
316        Args:
317            points:
318                ``N x 3`` point array.
319
320            uv:
321                ``N x 2`` pixel array aligned with ``points``.
322
323        Returns:
324            Filtered ``(points, uv)`` tuple. If Open3D reports no inliers, two
325            empty arrays are returned.
326
327        Notes:
328            Exceptions from Open3D fall back to the unfiltered inputs.
329        """
330        # 2) Statistical 3D outlier removal (keeps mapping by applying indices to uv)
331        #print(f"number of points before S3O {len(points)}")
332        try:
333            pc_clean = o3d.geometry.PointCloud()
334            pc_clean.points = o3d.utility.Vector3dVector(points)
335            pc_filtered, ind = pc_clean.remove_statistical_outlier(nb_neighbors=self.nb_neighbors, std_ratio=self.std_ratio)
336            ind = np.array(ind, dtype=int)
337            if ind.size == 0:
338                return np.empty((0, 3)), np.empty((0, 2))
339            points_2_3d_outlier_removal = points[ind]
340            uv_2_3d_outlier_removal = uv[ind]
341        except Exception:
342            points_2_3d_outlier_removal = points
343            uv_2_3d_outlier_removal = uv
344        if points_2_3d_outlier_removal.shape[0] == 0:
345            return np.empty((0, 3)), np.empty((0, 2))
346        return points_2_3d_outlier_removal, uv_2_3d_outlier_removal

348    def connected_components_filter(self, points, uv, W, H, floor_height):
349        """
350        Reject small or depth-incoherent point patches in image space.
351
352        The sparse UV samples are dilated to restore connectivity lost through
353        subsampling. Each connected component is gated by area, median depth,
354        floor-relative height, and median absolute depth deviation.
355
356        Args:
357            points:
358                ``N x 3`` camera-frame point array.
359
360            uv:
361                ``N x 2`` pixel coordinates paired with ``points``.
362
363            W:
364                Image width in pixels.
365
366            H:
367                Image height in pixels.
368
369            floor_height:
370                Estimated camera-frame floor Y coordinate, or ``None``.
371
372        Returns:
373            Tuple of retained points and their UV coordinates.
374
375        Notes:
376            On an unexpected processing error, the method returns the original
377            input arrays. When the generated mask has no foreground component,
378            it returns empty arrays.
379        """
380        #print(f"number of points before CC {len(points)}")
381        # --- Remove small patches in image space (connected components) ---
382        try:
383            # Defaults so visualization doesn't disappear if nothing is filtered
384            uv_5_remove_patches = uv
385            points_5_remove_patches = points
386
387            area_scale = self.subsample * self.subsample  # compensate for subsampling
388            # Depth-aware params
389            # be it corridor or local passability check, the depth range is small (like 0.5m for local and 2-3m for corridor). so using min_z and max_z for area scaling.
390            # because setting min_area_near and minimum_area_far has nothing to do with depth of points in corridor or local passability check. these are just area thresholds to remove small patches.
391            # because points that are about 0.5m to robot is considered as near points( always )
392            min_area_near = max(6, int(self.min_area_near / area_scale))
393            minimum_area_far = max(6, int(self.minimum_area_far / area_scale))
394
395            # Build 1px mask (no morphology and vectorized)
396            uv_int = np.round(uv).astype(int)
397            valid_uv = (
398                (uv_int[:, 0] >= 0) & (uv_int[:, 0] < W) &
399                (uv_int[:, 1] >= 0) & (uv_int[:, 1] < H)
400            )
401            mask = np.zeros((H, W), np.uint8)
402            mask[uv_int[valid_uv, 1], uv_int[valid_uv, 0]] = 255
403            #kernel_size = max(7, subsample*2+1)
404            kernel_size = self.subsample
405            #“fill” the sparse mask a bit to restore local connectivity  due to subsampling
406            #Use subsample itself as the kernel size so it scales automatically
407            mask = cv2.dilate(mask, np.ones((kernel_size, kernel_size), np.uint8), iterations=1)
408
409            num, labels, stats, _ = cv2.connectedComponentsWithStats(mask, connectivity=8)
410
411            if num <= 1:
412                return np.empty((0, 3)), np.empty((0, 2))
413
414            # Vectorized mapping: label lookup per UV pixel
415            lbl_values = labels[
416                np.clip(uv_int[valid_uv, 1], 0, H - 1),
417                np.clip(uv_int[valid_uv, 0], 0, W - 1)
418            ]
419
420            # Filter out background (label 0)
421            valid_labels = lbl_values > 0
422            lbl_values = lbl_values[valid_labels]
423            uv_valid = uv[valid_uv][valid_labels]
424            pts_valid = points[valid_uv][valid_labels]
425
426            # Precompute per-component area and median depth
427            unique_lbls, inverse_idx = np.unique(lbl_values, return_inverse=True)
428            areas = stats[unique_lbls, cv2.CC_STAT_AREA]
429
430            keep_mask = np.zeros(len(lbl_values), dtype=bool)
431
432            for i, lbl in enumerate(unique_lbls):
433                """
434                ####        visualization of connected components   #####
435                # Create mask for this component
436                comp_mask = (labels == lbl).astype(np.uint8)
437
438                # Create a visualization copy
439                vis = color_image.copy()
440
441                # Colorize the component (green overlay)
442                vis[comp_mask == 1] = (0, 0, 255)   # set component pixels to red
443
444                # Optionally, dim background to highlight the component better
445                background_mask = (comp_mask == 0)
446                vis[background_mask] = (vis[background_mask] * 0.3).astype(np.uint8)
447
448                # Add label text
449                x, y, w, h, area = stats[lbl]
450                cv2.putText(vis, f"ID:{lbl}, area:{area}", (x, y - 5),
451                            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,255,0), 1)
452
453                cv2.imshow(f"Component {lbl}", vis)
454                ####    visualization of connected components  end  ####    
455                """
456
457                label_mask = inverse_idx == i
458                area_px = areas[i]
459                z_vals = pts_valid[label_mask, 2]
460                z_vals = z_vals[~np.isnan(z_vals)]
461                if z_vals.size == 0:
462                    continue
463                z_med = np.median(z_vals)
464                
465                #z_eff = np.clip(z_med, self.min_z, self.max_z)
466
467                y_vals = pts_valid[label_mask, 1]
468                y_vals = y_vals[~np.isnan(y_vals)]
469                if y_vals.size == 0:
470                    continue
471                y_med = np.median(y_vals)
472                if floor_height is not None and y_med > floor_height + 0.1:
473                    continue
474
475                #there could be floor noisy points due to reflection which have high z value but small area. so these should be removed.
476                #depths above 3.5m are removed during back projection itself.but there can be noisy floor points with depth less than 3.5m but higher than actual floor depth.
477                z_mad = np.median(np.abs(z_vals - np.median(z_vals)))
478                # need to review if below code is needed or not. because remving points based on  depth variation can cause removal of valid points also.
479                if z_mad > (0.30 + 0.05 * (area_px / 100)):
480                    continue  # skip component with wildly varying depth
481                #area_thresh = base_area_at_1m / (z_eff * z_eff)
482                #using above line will not remove patches due to floor reflection. the points in floor near to robot will have high z_eff which will make area_thresh small and hence these patches will be kept instead of removing them.
483                #so use minimum_area_far as threshold for far points.
484
485                if (z_med < self.near_z and area_px >= min_area_near) or (area_px >= minimum_area_far):
486                    keep_mask[label_mask] = True
487
488            if np.any(keep_mask):
489                uv_5_remove_patches = uv_valid[keep_mask]
490                points_5_remove_patches = pts_valid[keep_mask]
491        except Exception:
492            # fall back without filtering on any error
493            uv_5_remove_patches = uv
494            points_5_remove_patches = points
495
496        print(f"number of points after CC {len(points_5_remove_patches)}")
497        return points_5_remove_patches, uv_5_remove_patches

537    def run(self, depth_image_in_meters, color_image, corridor_pts_input, corridor_uv_input, z_max):
538        """
539        Run the complete obstacle-filtering and clearance pipeline.
540
541        Stages execute in the following order: floor removal, normal filtering,
542        statistical outlier rejection, and image-space component filtering.
543        The nearest surviving Z value becomes the front clearance.
544
545        Args:
546            depth_image_in_meters:
547                Aligned ``H x W`` metric depth image.
548
549            color_image:
550                BGR image modified with filter-stage diagnostics.
551
552            corridor_pts_input:
553                Camera-frame 3D points already restricted to the corridor.
554
555            corridor_uv_input:
556                Pixels corresponding one-to-one with the input points.
557
558            z_max:
559                Far limit reported when no meaningful obstacle survives.
560
561        Returns:
562            Tuple ``(front_clearance, visualization, final_points)`` where
563            clearance is in meters and ``final_points`` contains the surviving
564            obstacle cloud.
565
566        Notes:
567            If no more than ``minimum_depth_points_after_filtering`` points
568            survive, the region is treated as clear up to ``z_max``.
569        """
570
571        H, W = depth_image_in_meters.shape
572        self.timer.start(f"check_if_passable--> main passability check")
573
574        self.timer.start(f"check_if_passable--> floor removal")
575        points_3_nofloor, uv_3_nofloor, floor_height = self.remove_floor_ransac(
576            corridor_pts_input, corridor_uv_input
577        )
578        self.timer.stop(f"check_if_passable--> floor removal")
579
580        self.timer.start(f"check_if_passable--> normal filtering")
581        points_4_normal, uv_4_normal = self.normal_filtering(points_3_nofloor, uv_3_nofloor)
582        self.timer.stop(f"check_if_passable--> normal filtering")
583
584        self.timer.start(f"check_if_passable--> 3D outlier removal")

349    def connected_components_filter(self, points, uv, W, H, floor_height):
350        """
351        Reject small or depth-incoherent point patches in image space.
352
353        The sparse UV samples are dilated to restore connectivity lost through
354        subsampling. Each connected component is gated by area, median depth,
355        floor-relative height, and median absolute depth deviation.
356
357        Args:
358            points:
359                ``N x 3`` camera-frame point array.
360
361            uv:
362                ``N x 2`` pixel coordinates paired with ``points``.
363
364            W:
365                Image width in pixels.
366
367            H:
368                Image height in pixels.
369
370            floor_height:
371                Estimated camera-frame floor Y coordinate, or ``None``.
372
373        Returns:
374            Tuple of retained points and their UV coordinates.
375
376        Notes:
377            On an unexpected processing error, the method returns the original
378            input arrays. When the generated mask has no foreground component,
379            it returns empty arrays.
380        """
381        #print(f"number of points before CC {len(points)}")
382        # --- Remove small patches in image space (connected components) ---
383        try:
384            # Defaults so visualization doesn't disappear if nothing is filtered
385            uv_5_remove_patches = uv
386            points_5_remove_patches = points
387
388            area_scale = self.subsample * self.subsample  # compensate for subsampling
389            # Depth-aware params
390            # be it corridor or local passability check, the depth range is small (like 0.5m for local and 2-3m for corridor). so using min_z and max_z for area scaling.
391            # because setting min_area_near and minimum_area_far has nothing to do with depth of points in corridor or local passability check. these are just area thresholds to remove small patches.
392            # because points that are about 0.5m to robot is considered as near points( always )
393            min_area_near = max(6, int(self.min_area_near / area_scale))
394            minimum_area_far = max(6, int(self.minimum_area_far / area_scale))
395
396            # Build 1px mask (no morphology and vectorized)
397            uv_int = np.round(uv).astype(int)
398            valid_uv = (
399                (uv_int[:, 0] >= 0) & (uv_int[:, 0] < W) &
400                (uv_int[:, 1] >= 0) & (uv_int[:, 1] < H)
401            )
402            mask = np.zeros((H, W), np.uint8)
403            mask[uv_int[valid_uv, 1], uv_int[valid_uv, 0]] = 255
404            #kernel_size = max(7, subsample*2+1)
405            kernel_size = self.subsample
406            #“fill” the sparse mask a bit to restore local connectivity  due to subsampling
407            #Use subsample itself as the kernel size so it scales automatically
408            mask = cv2.dilate(mask, np.ones((kernel_size, kernel_size), np.uint8), iterations=1)
409
410            num, labels, stats, _ = cv2.connectedComponentsWithStats(mask, connectivity=8)
411
412            if num <= 1:
413                return np.empty((0, 3)), np.empty((0, 2))
414
415            # Vectorized mapping: label lookup per UV pixel
416            lbl_values = labels[
417                np.clip(uv_int[valid_uv, 1], 0, H - 1),
418                np.clip(uv_int[valid_uv, 0], 0, W - 1)
419            ]
420
421            # Filter out background (label 0)
422            valid_labels = lbl_values > 0
423            lbl_values = lbl_values[valid_labels]
424            uv_valid = uv[valid_uv][valid_labels]
425            pts_valid = points[valid_uv][valid_labels]
426
427            # Precompute per-component area and median depth
428            unique_lbls, inverse_idx = np.unique(lbl_values, return_inverse=True)
429            areas = stats[unique_lbls, cv2.CC_STAT_AREA]
430
431            keep_mask = np.zeros(len(lbl_values), dtype=bool)
432
433            for i, lbl in enumerate(unique_lbls):
434                """
435                ####        visualization of connected components   #####
436                # Create mask for this component
437                comp_mask = (labels == lbl).astype(np.uint8)
438
439                # Create a visualization copy
440                vis = color_image.copy()
441
442                # Colorize the component (green overlay)
443                vis[comp_mask == 1] = (0, 0, 255)   # set component pixels to red
444
445                # Optionally, dim background to highlight the component better
446                background_mask = (comp_mask == 0)
447                vis[background_mask] = (vis[background_mask] * 0.3).astype(np.uint8)
448
449                # Add label text
450                x, y, w, h, area = stats[lbl]
451                cv2.putText(vis, f"ID:{lbl}, area:{area}", (x, y - 5),
452                            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,255,0), 1)
453
454                cv2.imshow(f"Component {lbl}", vis)
455                ####    visualization of connected components  end  ####    
456                """
457
458                label_mask = inverse_idx == i
459                area_px = areas[i]
460                z_vals = pts_valid[label_mask, 2]
461                z_vals = z_vals[~np.isnan(z_vals)]
462                if z_vals.size == 0:
463                    continue
464                z_med = np.median(z_vals)
465                
466                #z_eff = np.clip(z_med, self.min_z, self.max_z)
467
468                y_vals = pts_valid[label_mask, 1]
469                y_vals = y_vals[~np.isnan(y_vals)]
470                if y_vals.size == 0:
471                    continue
472                y_med = np.median(y_vals)
473                if floor_height is not None and y_med > floor_height + 0.1:
474                    continue
475
476                #there could be floor noisy points due to reflection which have high z value but small area. so these should be removed.
477                #depths above 3.5m are removed during back projection itself.but there can be noisy floor points with depth less than 3.5m but higher than actual floor depth.
478                z_mad = np.median(np.abs(z_vals - np.median(z_vals)))
479                # need to review if below code is needed or not. because remving points based on  depth variation can cause removal of valid points also.
480                if z_mad > (0.30 + 0.05 * (area_px / 100)):
481                    continue  # skip component with wildly varying depth
482                #area_thresh = base_area_at_1m / (z_eff * z_eff)
483                #using above line will not remove patches due to floor reflection. the points in floor near to robot will have high z_eff which will make area_thresh small and hence these patches will be kept instead of removing them.
484                #so use minimum_area_far as threshold for far points.
485
486                if (z_med < self.near_z and area_px >= min_area_near) or (area_px >= minimum_area_far):
487                    keep_mask[label_mask] = True
488
489            if np.any(keep_mask):
490                uv_5_remove_patches = uv_valid[keep_mask]
491                points_5_remove_patches = pts_valid[keep_mask]
492        except Exception:
493            # fall back without filtering on any error
494            uv_5_remove_patches = uv
495            points_5_remove_patches = points
496
497        print(f"number of points after CC {len(points_5_remove_patches)}")
498        return points_5_remove_patches, uv_5_remove_patches
