 15class TemporalPlaneTracker:
 16    """
 17    Stabilize a single plane hypothesis across consecutive frames.
 18
 19    New planes are associated with the smoothed hypothesis using normal-angle
 20    and distance gates. Associated measurements update an exponential moving
 21    average, while a bounded Boolean history applies separate confirmation and
 22    drop thresholds.
 23
 24    Attributes:
 25        alpha:
 26            Exponential moving-average weight of the newest measurement.
 27
 28        max_jump_deg:
 29            Maximum normal-angle change allowed for association.
 30
 31        max_jump_m:
 32            Maximum distance change allowed for association.
 33
 34        window:
 35            Maximum number of association decisions retained.
 36
 37        confirm_k:
 38            Associated-frame count required to confirm an unconfirmed track.
 39
 40        drop_k:
 41            Associated-frame count at or below which a confirmed track drops.
 42
 43        n_s:
 44            Smoothed unit plane normal, or ``None`` before initialization.
 45
 46        d_s:
 47            Smoothed plane distance in meters.
 48
 49        history:
 50            Recent Boolean association decisions.
 51
 52        confirmed:
 53            Current hysteretic confirmation state.
 54    """
 55
 56    def __init__(self):
 57        """
 58        Initialize temporal thresholds and an empty plane track.
 59
 60        Configuration is loaded from
 61        ``~plane_detector/temporal_smoothing``. No plane is considered
 62        confirmed until :meth:`update` has associated enough observations
 63        within the configured angular and distance jump limits.
 64
 65        Notes:
 66            The tracker starts without a smoothed normal or distance. Its
 67            history contains only association decisions made after the first
 68            candidate initializes the track.
 69        """
 70        ns = "~plane_detector/temporal_smoothing"
 71        self.alpha = rospy.get_param(f"{ns}/alpha", 0.3)
 72        self.max_jump_deg = rospy.get_param(f"{ns}/max_jump_deg", 12.0)
 73        self.max_jump_m = rospy.get_param(f"{ns}/max_jump_m", 0.4)
 74        self.window = int(rospy.get_param(f"{ns}/window", 10))
 75        self.confirm_k = int(rospy.get_param(f"{ns}/confirm_k", 6))
 76        self.drop_k = int(rospy.get_param(f"{ns}/drop_k", 2))
 77        self.n_s = None
 78        self.d_s = None
 79        self.history = []  # bool list of associated frames
 80        self.confirmed = False
 81
 82    def _normalize(self, n):
 83        """
 84        Normalize a vector without dividing by exact zero.
 85
 86        Args:
 87            n:
 88                Array-like vector.
 89
 90        Returns:
 91            NumPy vector divided by its norm plus a small numerical epsilon.
 92        """
 93        norm = np.linalg.norm(n)
 94        return n / (norm + 1e-12)
 95
 96    def _angle_deg(self, n1, n2):
 97        """
 98        Compute the unsigned angle between two unit vectors.
 99
100        Args:
101            n1:
102                First unit vector, or ``None``.
103
104            n2:
105                Second unit vector, or ``None``.
106
107        Returns:
108            Angle in degrees. Missing vectors return ``180.0``.
109        """
110        if n1 is None or n2 is None:
111            return 180.0
112
113        c = np.clip(float(np.dot(n1, n2)), -1.0, 1.0)
114        return np.degrees(np.arccos(c))
115
116    def _push(self, val: bool):
117        """
118        Append one association result to the bounded history.
119
120        Args:
121            val:
122                Whether the current measurement matched the tracked plane.
123        """
124        self.history.append(bool(val))
125        if len(self.history) > self.window:
126            self.history.pop(0)
127
128    def _update_confirmed(self):
129        """
130        Recompute confirmation state using hysteresis thresholds.
131
132        Notes:
133            An unconfirmed track needs at least ``confirm_k`` true entries,
134            while a confirmed track is retained until that count falls to
135            ``drop_k`` or below.
136        """
137        #counts the number of True in history. If >= confirm_k, set confirmed to True. If <= drop_k, set confirmed to False.
138        #self.history is a list of booleans. np.sum of False is 0, True is 1.
139        count_true = int(np.sum(self.history))
140        #hysteresis
141        if not self.confirmed and count_true >= self.confirm_k:
142            self.confirmed = True
143        elif self.confirmed and count_true <= self.drop_k:
144            self.confirmed = False
145
146    def update(self, n_t, d_t):
147        """
148        Associate and smooth one plane measurement.
149
150        Args:
151            n_t:
152                Current plane normal. It is normalized internally; ``None``
153                records a missed association.
154
155            d_t:
156                Current plane distance in meters. Nonfinite values are treated
157                as a missed association.
158
159        Returns:
160            Tuple ``(confirmed, smoothed_normal, smoothed_distance)``.
161
162        Notes:
163            A valid first measurement initializes the smoothed hypothesis.
164            Later measurements update it only when both association gates pass.
165        """
166        if n_t is None or not np.isfinite(d_t):
167            self._push(False)
168            self._update_confirmed()
169            return self.confirmed, self.n_s, self.d_s
170
171        n_t = self._normalize(n_t)
172        d_t = float(d_t)
173
174        # initialize
175        if self.n_s is None:
176            self.n_s = n_t
177            self.d_s = d_t
178            self._push(True)
179            self._update_confirmed()
180            return self.confirmed, self.n_s, self.d_s
181
182        angle = self._angle_deg(self.n_s, n_t)
183        jump = abs(d_t - self.d_s)
184        associated = (angle <= self.max_jump_deg) and (jump <= self.max_jump_m)
185        self._push(associated)
186        if associated:
187            # EMA smoothing
188            n_blend = (1.0 - self.alpha) * self.n_s + self.alpha * n_t
189            self.n_s = self._normalize(n_blend)
190            self.d_s = (1.0 - self.alpha) * self.d_s + self.alpha * d_t
191        self._update_confirmed()
192        return self.confirmed, self.n_s, self.d_s

326    def find_vertical_planes(self,points):
327        """
328        Iteratively segment front-facing planes from a point cloud.
329
330        Args:
331            points:
332                ``N x 3`` filtered camera-frame point array.
333
334        Returns:
335            List of ``(plane_model, original_indices, inlier_points)`` tuples
336            for accepted planes.
337
338        Notes:
339            In this camera convention, a door plane faces the optical axis and
340            therefore has ``abs(normal_z)`` close to one. Inliers are removed
341            after every fit so later iterations can find another plane.
342        """
343
344        remaining_points = points.copy()
345        remaining_indices = np.arange(points.shape[0])  # Track original indices
346        found_vertical_planes = []
347
348
349
350        for _ in range(self.max_planes):
351            if len(remaining_points) < self.ransac_n:
352                break
353
354            pc = o3d.geometry.PointCloud()
355            pc.points = o3d.utility.Vector3dVector(remaining_points)
356            plane_model, inliers = pc.segment_plane(distance_threshold=self.ransac_distance_threshold,
357                                                    ransac_n=self.ransac_n,
358                                                    num_iterations=self.ransac_num_iterations)
359
360            inliers = np.array(inliers, dtype=np.int64)
361            if len(inliers) < self.min_inliers:
362                break
363
364            orig_inlier_indices = remaining_indices[inliers]
365            orig_inlier_points = remaining_points[inliers]
366
367
368            # Check if the plane is vertical
369            normal = np.array(plane_model[:3])
370            normal = normal / np.linalg.norm(normal)
371            #print("Detected plane normal:", normal)
372            if abs(abs(normal[2]) - 1.0) < self.vertical_tol:
373                # Save vertical plane
374                found_vertical_planes.append((plane_model, orig_inlier_indices, orig_inlier_points))
375                print(f"Found vertical plane with {len(inliers)} inliers.")
376            else:
377                print("Detected plane is not vertical; skipping.")
378
379
380            mask = np.ones(len(remaining_points), dtype=bool)
381            mask[inliers] = False
382
383            remaining_points = remaining_points[mask]
384            remaining_indices = remaining_indices[mask]
385        return found_vertical_planes

434    def voxel_downsample(self, points, uv):
435        """
436        Keep one point per voxel while preserving point-to-pixel indices.
437
438        Args:
439            points:
440                ``N x 3`` point array.
441
442            uv:
443                ``N x 2`` matching pixel array.
444
445        Returns:
446            Tuple of downsampled points and corresponding pixels.
447
448        Notes:
449            The first source index found in each voxel is retained.
450        """
451        # Voxel downsampling
452        # Reduces redundant neighboring points while preserving
453        # metal frame edges and door boundaries. This balances
454        # point density for RANSAC without losing structure.
455        #add size heuristics and inliner density based on high depth change(glass)
456
457        vox_coords = np.floor(points / self.voxel_size).astype(np.int64)
458        _, unique_idx = np.unique(vox_coords, axis=0, return_index=True)
459        unique_idx = np.sort(np.array(unique_idx, dtype=np.int64))
460        points = points[unique_idx]
461        uv = uv[unique_idx]
462        return points, uv

464    def normal_filter(self, points, uv):
465        """
466        Keep surfaces whose normals face roughly along the camera Z axis.
467
468        Args:
469            points:
470                Voxel-downsampled ``N x 3`` point array.
471
472            uv:
473                ``N x 2`` pixels paired with ``points``.
474
475        Returns:
476            Tuple of normal-filtered points and matching pixels.
477
478        Notes:
479            Accepted normals satisfy all configured X, Y, and Z component
480            thresholds.
481        """
482        pc = o3d.geometry.PointCloud()
483        pc.points = o3d.utility.Vector3dVector(points)
484
485        # Normal estimation
486        pc.estimate_normals(
487            search_param=o3d.geometry.KDTreeSearchParamHybrid(radius=self.normal_radius, max_nn=self.normal_max_nn)
488        )
489        normals = np.asarray(pc.normals)
490
491        # Normal filtering
492        z_axis_mask = (np.abs(normals[:, 0]) < self.nx_thr) & (np.abs(normals[:, 1]) < self.ny_thr) & (np.abs(normals[:, 2]) >= self.nz_min)
493        keep_idx = np.where(z_axis_mask)[0]
494
495        pc = pc.select_by_index(keep_idx)
496        points = np.asarray(pc.points)
497        uv = uv[keep_idx]
498        return points, uv

501    def check_plane_size(self, inlier_points, plane_model):
502        """
503        Check whether a plane has door-like physical width and height.
504
505        Args:
506            inlier_points:
507                ``N x 3`` RANSAC inlier points.
508
509            plane_model:
510                Coefficients ``(a, b, c, d)`` for
511                ``a*x + b*y + c*z + d = 0``.
512
513        Returns:
514            Four robust 3D corners when the plane satisfies the configured
515            dimensions, otherwise ``None``.
516
517        Notes:
518            X and Y bounds use configured percentiles rather than raw extrema.
519            Z for each corner is solved from the fitted plane equation.
520        """
521        if inlier_points.shape[0] == 0:
522            return None
523        x_min = np.percentile(inlier_points[:, 0], self.lower_pct)
524        x_max = np.percentile(inlier_points[:, 0], self.upper_pct)
525        y_min = np.percentile(inlier_points[:, 1], self.lower_pct)
526        y_max = np.percentile(inlier_points[:, 1], self.upper_pct)
527
528        width = x_max - x_min
529        height = y_max - y_min
530        if width < self.min_width_m or height < self.min_height_m:
531            return None  # Plane too small to consider
532
533        corners_3d = np.array([
534            [x_min, y_min, 0],
535            [x_max, y_min, 0],
536            [x_max, y_max, 0],
537            [x_min, y_max, 0]
538        ])
539
540
541        a, b, c, d = plane_model
542        for i in range(4):
543            x, y, _ = corners_3d[i]
544            z = -(a*x + b*y + d) / c
545            corners_3d[i, 2] = z
546
547        return corners_3d

549    def find_plane_metrics(self, plane_model, depth_image_in_meters, inlier_indices, plane_corners_2d, mask):
550        """
551        Measure depth spread, holes, density, distance, and plane normal.
552
553        Args:
554            plane_model:
555                Plane coefficients ``(a, b, c, d)``.
556
557            depth_image_in_meters:
558                Aligned metric depth image.
559
560            inlier_indices:
561                Source indices belonging to the selected plane.
562
563            plane_corners_2d:
564                Projected image-space plane polygon.
565
566            mask:
567                Caller-provided uint8 image mask modified in place.
568
569        Returns:
570            Dictionary containing robust depth statistics, hole fraction,
571            inlier density, origin-to-plane distance, and a unit normal.
572
573        Notes:
574            Depth values outside the 5th-to-95th percentile interval are
575            discarded before calculating mean and standard deviation.
576        """
577        pts = np.array(plane_corners_2d, dtype=np.int32).reshape((-1, 1, 2))
578        cv2.fillPoly(mask, [pts], 255)
579
580        fraction_of_holes_in_plane = (np.sum((mask == 255) & ((depth_image_in_meters == 0) | ~np.isfinite(depth_image_in_meters)))) / (np.sum(mask == 255) + 1e-9)
581        inlier_density = len(inlier_indices) / (np.sum(mask == 255) + 1e-9)
582
583        depths_in_plane = depth_image_in_meters[mask == 255]
584        depths_in_plane = depths_in_plane[np.isfinite(depths_in_plane) & (depths_in_plane > 0)]
585        depth_min_p_5 = np.percentile(depths_in_plane, 5)
586        depth_max_p_95 = np.percentile(depths_in_plane, 95)
587        depths_in_plane = depths_in_plane[(depths_in_plane >= depth_min_p_5) & (depths_in_plane <= depth_max_p_95)]
588
589
590        a, b, c, d = plane_model
591        plane_norm = np.sqrt(a*a + b*b + c*c)
592        distance_m = abs(d) / (plane_norm + 1e-12)
593        plane_norm_vector = np.array([a, b, c], dtype=np.float32) / plane_norm
594
595        depth_mean = np.mean(depths_in_plane)
596        depth_std = np.std(depths_in_plane)
597        normalized_std = depth_std / (depth_mean + 1e-9)
598
599        return {
600            "depth_mean": depth_mean,
601            "depth_std": depth_std,
602            "normalized_std": normalized_std,
603            "depth_min": depth_min_p_5,
604            "depth_max": depth_max_p_95,
605            "fraction_of_holes_in_plane": fraction_of_holes_in_plane,
606            "inlier_density": inlier_density,
607            "distance_m": distance_m,
608            "plane_norm_vector" : plane_norm_vector
609
610        }

661
662        Candidates must face the camera, satisfy physical size and distance
663        gates, and remain geometrically consistent across the tracker's recent
664        history. The color image receives candidate and confirmed overlays.
665
666        Args:
667            color_image:
668                BGR image modified with RANSAC inliers and confirmed outline.
669
670            depth_image_in_meters:
671                Aligned ``H x W`` metric depth image.
672

674                Horizontal focal length in pixels.
675
676            fy:
677                Vertical focal length in pixels.
678

680                Horizontal principal point in pixels.
681
682            cy:
683                Vertical principal point in pixels.
684

686            A dictionary containing the confirmed plane model, derived metrics,
687            inlier points, and flags distinguishing absent from unconfirmed
688            candidates.

691            ``had_candidates`` is true when geometry was found but rejected by
692            distance or not yet confirmed. A confirmed result is selected by
693            maximum inlier count among candidates inside the distance gate.
694        """
695
696
697        all_points, uv, valid_mask = backproject_depth_to_points(
698            depth_image_in_meters, fx, fy, cx, cy,
699            self.max_depth_backprojection, self.min_depth_backprojection,
700            self.subsample
701        )
702
703
704        if all_points.shape[0] == 0:
705            return {"plane_model": None, "plane_metrics": None, "inlier_points": None,"had_candidates": False, "confirmed": False}
706
707
708        points_after_voxel_downsample, uv_after_voxel_downsample = self.voxel_downsample(all_points, uv)
709        #points_after_voxel_downsample, uv_after_voxel_downsample = all_points, uv
710        points_after_normal_filtering, uv_after_normal_filtering = self.normal_filter(points_after_voxel_downsample, uv_after_voxel_downsample)
711
712
713        # RANSAC vertical planes
714        found_vertical_planes = self.find_vertical_planes(points_after_normal_filtering)
715        if not found_vertical_planes:
716            return {"plane_model": None, "plane_metrics": None,"inlier_points": None, "had_candidates": False, "confirmed": False}
717
718        self.highlight_planes_on_image(
719            color_image,uv_after_normal_filtering,
720            found_vertical_planes)
721
722        # Build candidates with size gating; defer drawing until confirmed
723        candidates = []
724        for plane_model, inlier_indices, inlier_points in found_vertical_planes:
725            if inlier_indices.shape[0] == 0:
726                continue
727
728            plane_corners_3d = self.check_plane_size(inlier_points, plane_model)
729            # if the plane is too small, skip drawing and continue
730            if plane_corners_3d is None:
731                continue
732            a, b, c, d = plane_model
733            plane_norm = np.sqrt(a*a + b*b + c*c)
734            if plane_norm <= 1e-12:
735                continue
736            n = np.array([a, b, c], dtype=np.float32) / plane_norm
737            distance_m = abs(d) / plane_norm
738            candidates.append({
739                "plane_model": plane_model,
740                "inliers": inlier_indices,
741                "inlier_points": inlier_points,
742                "corners_3d": plane_corners_3d,
743                "normal": n,
744                "distance_m": float(distance_m),
745            })
746
747        if not candidates:
748            return {"plane_model": None, "plane_metrics": None, "inlier_points": None,"had_candidates": False, "confirmed": False}
749
750        # Keep only planes within the reference distance threshold
751        filtered = [c for c in candidates if abs(c["distance_m"] - self.reference_door_distance_m) <= self.distance_range_m]
752        if not filtered:
753            # We had candidates, but none within desired distance
754            return {"plane_model": None, "plane_metrics": None, "inlier_points": None, "had_candidates": True, "confirmed": False}
755        # Pick the strongest by inlier count among filtered
756        # this is important to avoid jumping between multiple planes. because sometimes, points that belongs to one plane is segmented into two planes by RANSAC.
757        # observed in testing glass door with walls on both sides.
758        chosen = max(filtered, key=lambda c: len(c["inliers"]))
759
760        # Update temporal tracker; only proceed once confirmed
761        confirmed, n_s, d_s = self.tracker.update(chosen["normal"], chosen["distance_m"])
762        if not confirmed:
763            return {"plane_model": None, "plane_metrics": None, "inlier_points": None, "had_candidates": True, "confirmed": False}
764
765        # Draw outline and compute metrics for confirmed plane. only after temporal smoothing.
766        plane_corners_2d = self.draw_plane_outline_on_image(
767            color_image, chosen["corners_3d"], fx, fy, cx, cy)
768
769        mask = np.zeros(color_image.shape[:2], dtype=np.uint8)
770        plane_metrics = self.find_plane_metrics(
771            chosen["plane_model"], depth_image_in_meters, chosen["inliers"], plane_corners_2d, mask)
772
773        result = {
774            "plane_model": chosen["plane_model"],
775            "plane_metrics": plane_metrics,
776            "inlier_points" : chosen["inlier_points"],
777            "had_candidates": True,
778            "confirmed": True
779        }
780
781        return result

129    def _update_confirmed(self):

140        count_true = int(np.sum(self.history))

152            n_t:

154                records a missed association.

552        Measure depth spread, holes, density, distance, and plane normal.
553
554        Args:
555            plane_model:
556                Plane coefficients ``(a, b, c, d)``.

718

720            color_image,uv_after_normal_filtering,
721            found_vertical_planes)
722
723        # Build candidates with size gating; defer drawing until confirmed
724        candidates = []
725        for plane_model, inlier_indices, inlier_points in found_vertical_planes:
726            if inlier_indices.shape[0] == 0:
727                continue
728
729            plane_corners_3d = self.check_plane_size(inlier_points, plane_model)
730            # if the plane is too small, skip drawing and continue
731            if plane_corners_3d is None:
732                continue
733            a, b, c, d = plane_model
734            plane_norm = np.sqrt(a*a + b*b + c*c)
735            if plane_norm <= 1e-12:
736                continue
737            n = np.array([a, b, c], dtype=np.float32) / plane_norm
738            distance_m = abs(d) / plane_norm
739            candidates.append({
740                "plane_model": plane_model,
741                "inliers": inlier_indices,
742                "inlier_points": inlier_points,
743                "corners_3d": plane_corners_3d,
744                "normal": n,
745                "distance_m": float(distance_m),
746            })
747
748        if not candidates:
749            return {"plane_model": None, "plane_metrics": None, "inlier_points": None,"had_candidates": False, "confirmed": False}
750
751        # Keep only planes within the reference distance threshold
752        filtered = [c for c in candidates if abs(c["distance_m"] - self.reference_door_distance_m) <= self.distance_range_m]
753        if not filtered:
754            # We had candidates, but none within desired distance
755            return {"plane_model": None, "plane_metrics": None, "inlier_points": None, "had_candidates": True, "confirmed": False}
756        # Pick the strongest by inlier count among filtered
757        # this is important to avoid jumping between multiple planes. because sometimes, points that belongs to one plane is segmented into two planes by RANSAC.
758        # observed in testing glass door with walls on both sides.
759        chosen = max(filtered, key=lambda c: len(c["inliers"]))
760
761        # Update temporal tracker; only proceed once confirmed
762        confirmed, n_s, d_s = self.tracker.update(chosen["normal"], chosen["distance_m"])
763        if not confirmed:
764            return {"plane_model": None, "plane_metrics": None, "inlier_points": None, "had_candidates": True, "confirmed": False}
765
766        # Draw outline and compute metrics for confirmed plane. only after temporal smoothing.
767        plane_corners_2d = self.draw_plane_outline_on_image(
768            color_image, chosen["corners_3d"], fx, fy, cx, cy)
769
770        mask = np.zeros(color_image.shape[:2], dtype=np.uint8)
771        plane_metrics = self.find_plane_metrics(
772            chosen["plane_model"], depth_image_in_meters, chosen["inliers"], plane_corners_2d, mask)
773
774        result = {
775            "plane_model": chosen["plane_model"],
776            "plane_metrics": plane_metrics,
777            "inlier_points" : chosen["inlier_points"],
778            "had_candidates": True,
779            "confirmed": True
780        }
781
782        return result
