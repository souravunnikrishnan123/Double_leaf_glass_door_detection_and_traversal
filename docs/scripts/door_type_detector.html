 56    def estimate_glass_and_frame_widths(self,
 57        inlier_points
 58    ):
 59        """
 60        Estimate glass and frame widths from one confirmed plane's inliers.
 61
 62        The key idea:
 63        A true frame supports the plane over MOST of the door height.
 64        Glass supports the plane only locally (bottom / sides).
 65
 66        Args:
 67            inlier_points:
 68                ``N x 3`` camera-frame XYZ coordinates of RANSAC inliers.
 69
 70        Returns:
 71            Dictionary containing metric bins, per-bin labels, consolidated
 72            segments, selected frame/glass widths, and selected segment
 73            indices. Returns ``None`` when the robust door height is not
 74            positive.
 75
 76        Notes:
 77            Door extents use the 2nd and 98th percentiles to limit isolated
 78            points. Final selection prefers a frame flanked by valid glass; if
 79            unavailable, a single valid glass segment may be paired with an
 80            adjacent frame.
 81        """
 82
 83        # Horizontal (width) coordinate
 84        xs = inlier_points[:, 0]
 85
 86        # Vertical (height) coordinate
 87        ys = inlier_points[:,1]  # Y in world coords is already vertical
 88
 89        # Total door height supported by inliers
 90        door_y_min = np.percentile(ys, 2)
 91        door_y_max = np.percentile(ys, 98)
 92
 93        door_x_min = np.percentile(xs, 2)
 94        door_x_max = np.percentile(xs, 98)
 95
 96        door_height = door_y_max - door_y_min
 97        if door_height <= 0:
 98            return None
 99
100        # ------------------------------------------------------------
101        # STEP 3: Create bins along door width (PHYSICAL bins)
102        # ------------------------------------------------------------
103
104
105        bin_edges = np.arange(door_x_min, door_x_max + self.bin_width, self.bin_width)
106
107        bin_labels = []
108        bins = []
109
110        # ------------------------------------------------------------
111        # STEP 4: Classify each bin using VERTICAL SUPPORT
112        # ------------------------------------------------------------
113
114        for i in range(len(bin_edges) - 1):
115            x0, x1 = bin_edges[i], bin_edges[i + 1]
116            bins.append((x0, x1))
117
118            # Inlier points whose horizontal coordinate falls in this bin
119            in_bin = (xs >= x0) & (xs < x1)
120
121            if np.count_nonzero(in_bin) < self.min_points_per_bin:
122                # No or very few inliers → no vertical support → glass
123                bin_labels.append("glass")
124                continue
125
126            y_vals = ys[in_bin]
127
128
129            vertical_support_ratio = self.compute_vertical_occupancy_ratio(
130                                        y_vals=y_vals,
131                                        door_y_min=door_y_min,
132                                        door_y_max=door_y_max
133                                    )
134
135            if vertical_support_ratio >= self.min_vertical_support:
136                bin_labels.append("frame")
137            else:
138                bin_labels.append("glass")
139
140        # ------------------------------------------------------------
141        # STEP 5: Merge consecutive bins into segments
142        # ------------------------------------------------------------
143
144        segments = []
145        start_idx = 0
146        current_label = bin_labels[0]
147
148        for i in range(1, len(bin_labels)):
149            if bin_labels[i] != current_label:
150                segments.append((
151                    bin_edges[start_idx],
152                    bin_edges[i],
153                    current_label
154                ))
155                start_idx = i
156                current_label = bin_labels[i]
157
158        # Add final segment
159        segments.append((
160            bin_edges[start_idx],
161            bin_edges[len(bin_labels)],
162            current_label
163        ))
164
165        # ------------------------------------------------------------
166        # STEP 6: Extract glass and frame widths
167        # ------------------------------------------------------------
168
169        glass_widths_m = []
170        frame_widths_m = []
171
172        for x0, x1, label in segments:
173            width = x1 - x0
174            if label == "glass":
175                glass_widths_m.append(width)
176            else:
177                frame_widths_m.append(width)
178
179        merged_segments = self.consolidate_door_segments(segments)
180
181        # ------------------------------------------------------------
182        # STEP 7: Select final frame/glass widths per adjacency rules
183        # ------------------------------------------------------------
184        final_frame_width_m = None
185        final_glass_width_m = None
186        final_frame_index = None
187        final_glass_index = None
188
189        if merged_segments and len(merged_segments) >= 1:
190            widths = [x1 - x0 for (x0, x1, _) in merged_segments]
191            labels = [label for (_, _, label) in merged_segments]
192
193            # Filter glass segments by max width constraint
194            glass_ok = set([i for i, label in enumerate(labels)
195                            if label == "glass" and widths[i] <= self.max_glass_width])
196
197            # Case A: a frame flanked by glass on both sides
198            candidate_frames = []
199            for i in range(len(merged_segments)):
200                if labels[i] != "frame":
201                    continue
202                left_ok = (i - 1) in glass_ok if i - 1 >= 0 and labels[i - 1] == "glass" else False
203                right_ok = (i + 1) in glass_ok if i + 1 < len(merged_segments) and labels[i + 1] == "glass" else False
204                if left_ok and right_ok:
205                    candidate_frames.append(i)
206
207            if candidate_frames:
208                # Choose the widest frame among candidates
209                i_best = max(candidate_frames, key=lambda k: widths[k])
210                final_frame_width_m = widths[i_best]
211                final_frame_index = i_best
212                final_glass_width_m = min(widths[i_best - 1], widths[i_best + 1])
213                final_glass_index = i_best - 1 if widths[i_best - 1] <= widths[i_best + 1] else i_best + 1
214            else:
215                # Case B: exactly one valid glass segment, use adjacent frames
216                if len(glass_ok) == 1:
217                    g_idx = list(glass_ok)[0]
218                    final_glass_width_m = widths[g_idx]
219                    final_glass_index = g_idx
220
221                    adj_frame_widths = []
222                    if g_idx - 1 >= 0 and labels[g_idx - 1] == "frame":
223                        adj_frame_widths.append(widths[g_idx - 1])
224
225                    if g_idx + 1 < len(merged_segments) and labels[g_idx + 1] == "frame":
226                        adj_frame_widths.append(widths[g_idx + 1])
227                    
228                    if adj_frame_widths:
229                        final_frame_width_m = min(adj_frame_widths)
230                        final_frame_index = g_idx - 1 if final_frame_width_m == widths[g_idx - 1] else g_idx + 1
231
232        result = {
233            "bins": bins,
234            "bin_labels": bin_labels,
235            "segments": merged_segments,
236            "final_frame_width_m": final_frame_width_m,
237            "final_glass_width_m": final_glass_width_m,
238            "final_frame_index": final_frame_index,
239            "final_glass_index": final_glass_index
240        }
241
242        return result

246    def compute_vertical_occupancy_ratio(self,
247        y_vals,
248        door_y_min,
249        door_y_max
250    ):
251        """
252        Measure how much of the door height is supported in one width bin.
253
254        Args:
255            y_vals:
256                Camera-frame Y coordinates belonging to one horizontal bin.
257
258            door_y_min:
259                Robust lower bound of the full door extent.
260
261            door_y_max:
262                Robust upper bound of the full door extent.
263
264        Returns:
265            Occupied vertical-slice fraction in the interval ``[0, 1]``.
266
267        Notes:
268            A slice is occupied only when it contains at least
269            ``min_points_per_vertical_slice`` inliers.
270        """
271
272
273        # Divide the FULL door height into equal slices
274        slice_edges = np.linspace(
275            door_y_min,
276            door_y_max,
277            self.num_vertical_slices + 1
278        )
279
280        occupied = 0
281
282        for i in range(self.num_vertical_slices):
283            y0 = slice_edges[i]
284            y1 = slice_edges[i + 1]
285
286            # Count inliers in this vertical slice
287            in_slice = (y_vals >= y0) & (y_vals < y1)
288
289            if np.count_nonzero(in_slice) >= self.min_points_per_vertical_slice:
290                occupied += 1
291
292        return occupied / self.num_vertical_slices

295    def consolidate_door_segments(self,
296    segments):
297        """
298        Remove tiny glass runs and merge adjacent equal-label segments.
299
300        Args:
301            segments:
302                Sequence of ``(x_start, x_end, label)`` metric segments.
303
304        Returns:
305            Consolidated segment list. If every input segment is removed, the
306            current implementation returns ``([], [])``.
307
308        Notes:
309            Frame segments are retained regardless of width. Glass segments
310            narrower than ``min_glass_width`` are discarded before merging.
311        """
312
313        # --------------------------------------------------
314        # STEP 1: Remove tiny GLASS segments (keep all frames)
315        # --------------------------------------------------
316        filtered = []
317        for seg in segments:
318            x0, x1, label = seg
319            width = x1 - x0
320            if label != "glass" or width >= self.min_glass_width:
321                filtered.append(seg)
322
323        if not filtered:
324            return [], []
325
326        # --------------------------------------------------
327        # STEP 2: Merge adjacent segments with same label
328        # --------------------------------------------------
329        merged = [filtered[0]]
330
331        for x0, x1, label in filtered[1:]:
332            px0, px1, plabel = merged[-1]
333
334            if label == plabel:
335                # Extend previous segment
336                merged[-1] = (px0, x1, label)
337            else:
338                merged.append((x0, x1, label))
339
340
341        return merged

348        bin_edges,
349        bin_labels,
350        segments,
351        final_glass_index,
352        final_frame_index,
353        fx, cx,
354        color_image
355    ):
356        """
357        Draw width-bin classifications and selected segments.
358
359        Args:
360            inlier_points:
361                Plane inliers used to fit the world-X to image-u mapping.
362
363            bin_edges:
364                Iterable of metric ``(x_start, x_end)`` bins.
365
366            bin_labels:
367                ``"frame"`` or ``"glass"`` label for each bin.
368
369            segments:
370                Consolidated ``(x_start, x_end, label)`` segments.
371
372            final_glass_index:
373                Index of the selected glass segment, or ``None``.
374
375            final_frame_index:
376                Index of the selected frame segment, or ``None``.
377
378            fx:
379                Horizontal focal length in pixels.
380
381            cx:
382                Horizontal principal point in pixels.
383
384            color_image:
385                Source BGR image.
386
387        Returns:
388            Annotated BGR image copy suitable for ROS publication.
389
390        Notes:
391            Per-bin translucent colors show the raw classification. Thick
392            outlines and text identify consolidated and final segments.
393        """
394
395        vis = color_image.copy()
396        H, W, _ = vis.shape
397
398        # --------------------------------------------------
399        # STEP 1: Map world X → image u (pixels)
400        # --------------------------------------------------
401        a, b = self.map_world_x_to_image_u(inlier_points, fx, cx)
402
403        def world_x_to_img_u(x):
404            """
405            Convert one metric X coordinate using the fitted linear mapping.
406
407            Args:
408                x:
409                    Camera-frame X coordinate in meters.
410
411            Returns:
412                Approximate horizontal image coordinate as an integer.
413            """
414            return int(a * x + b)
415
416        # --------------------------------------------------
417        # STEP 3: Draw per-bin visualization
418        # --------------------------------------------------
419        for (x0, x1), label in zip(bin_edges, bin_labels):
420            u0 = world_x_to_img_u(x0)
421            u1 = world_x_to_img_u(x1)
422
423            u0, u1 = np.clip([u0, u1], 0, W - 1)
424
425            if label == "frame":
426                color = (255, 0, 0)     # blue
427            else:
428                color = (0, 255, 255)   # yellow
429
430            overlay = vis.copy()
431            cv2.rectangle(
432                overlay,
433                (u0, 0),
434                (u1, H),
435                color,
436                -1
437            )
438            vis = cv2.addWeighted(overlay, 0.20, vis, 0.80, 0)
439
440            # Bin boundary
441            cv2.line(vis, (u0, 0), (u0, H), (0, 0, 255), 1)
442
443        # --------------------------------------------------
444        # STEP 4: Draw consolidated segments (thick boxes)
445        # --------------------------------------------------
446        for i, (x0, x1, label) in enumerate(segments):
447            u0 = world_x_to_img_u(x0)
448            u1 = world_x_to_img_u(x1)
449
450            u0, u1 = np.clip([u0, u1], 0, W - 1)
451
452            width_cm = (x1 - x0) * 100.0
453
454            if label == "frame":
455                if i == final_frame_index:
456                    color = (0, 0, 255)   # red
457                    text = f"Final Frame {width_cm:.1f} cm"
458                else:
459                    color = (0, 255, 0)   # green
460                    text = f"Frame {width_cm:.1f} cm"
461            else:
462                if i == final_glass_index:
463                    color = (255, 0, 0)   # blue
464                    text = f"Final Glass {width_cm:.1f} cm"
465                else:
466                    color = (0, 165, 255) # orange
467                    text = f"Glass {width_cm:.1f} cm"
468
469            cv2.rectangle(vis, (u0, 5), (u1, H - 5), color, 3)
470
471            cv2.putText(
472                vis,
473                text,
474                (u0 + 5, 30),
475                cv2.FONT_HERSHEY_SIMPLEX,
476                0.6,
477                color,
478                2,
479                cv2.LINE_AA
480            )
481
482        return vis

484    def map_world_x_to_image_u(self, inlier_points, fx, cx):
485        """
486        Fit a linear mapping from camera-frame X to horizontal image pixels.
487
488        Args:
489            inlier_points:
490                ``N x 3`` camera-frame point array.
491
492            fx:
493                Horizontal focal length in pixels.
494
495            cx:
496                Horizontal principal point in pixels.
497
498        Returns:
499            Tuple ``(slope, intercept)`` for ``u ≈ slope * X + intercept``.
500
501        Notes:
502            Each source point is first projected with its own Z value, then a
503            least-squares linear model is fitted against X.
504        """
505
506        xs = inlier_points[:, 0]
507        zs = inlier_points[:, 2]
508
509        # Project points to image u coordinate
510        us = (xs * fx / zs) + cx
511
512        # Fit linear mapping: u ≈ a * X + b
513        a, b = np.polyfit(xs, us, 1)
514        return a, b

485    def map_world_x_to_image_u(self, inlier_points, fx, cx):
486        """
487        Fit a linear mapping from camera-frame X to horizontal image pixels.
488
489        Args:
490            inlier_points:
491                ``N x 3`` camera-frame point array.
492
493            fx:
494                Horizontal focal length in pixels.
495
496            cx:
497                Horizontal principal point in pixels.
498
499        Returns:
500            Tuple ``(slope, intercept)`` for ``u ≈ slope * X + intercept``.
501
502        Notes:
503            Each source point is first projected with its own Z value, then a
504            least-squares linear model is fitted against X.
505        """
506
507        xs = inlier_points[:, 0]
508        zs = inlier_points[:, 2]
509
510        # Project points to image u coordinate
511        us = (xs * fx / zs) + cx
512
513        # Fit linear mapping: u ≈ a * X + b
514        a, b = np.polyfit(xs, us, 1)
515        return a, b

486        """
487        Fit a linear mapping from camera-frame X to horizontal image pixels.
488
489        Args:
490            inlier_points:
491                ``N x 3`` camera-frame point array.
492
493            fx:
494                Horizontal focal length in pixels.
495

497                Horizontal principal point in pixels.
498

500            Tuple ``(slope, intercept)`` for ``u ≈ slope * X + intercept``.
501
502        Notes:
503            Each source point is first projected with its own Z value, then a

505        """
506
507        xs = inlier_points[:, 0]

509
510        # Project points to image u coordinate

513        # Fit linear mapping: u ≈ a * X + b
514        a, b = np.polyfit(xs, us, 1)
